cmake_minimum_required(VERSION 3.16)
project(hetero_fleet_navigation)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(behaviortree_cpp REQUIRED)
add_library(hetero_stable_route_bt SHARED src/stable_replan.cpp)
target_compile_features(hetero_stable_route_bt PUBLIC cxx_std_17)
target_compile_definitions(hetero_stable_route_bt PRIVATE BT_PLUGIN_EXPORT)
target_link_libraries(hetero_stable_route_bt behaviortree_cpp::behaviortree_cpp)
ament_target_dependencies(hetero_stable_route_bt rclcpp nav_msgs)
install(TARGETS hetero_stable_route_bt LIBRARY DESTINATION lib)
ament_export_libraries(hetero_stable_route_bt)
if(BUILD_TESTING)
 enable_testing()
 add_executable(stable_route_test test/stable_route_test.cpp)
 target_link_libraries(stable_route_test behaviortree_cpp::behaviortree_cpp)
 ament_target_dependencies(stable_route_test rclcpp nav_msgs)
 add_test(NAME stable_route_behavior COMMAND stable_route_test $<TARGET_FILE:hetero_stable_route_bt>)
endif()
ament_package()
